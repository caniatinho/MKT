async function saveData(){

function render(){

  const dashboard = document.getElementById("dashboard");

  dashboard.innerHTML = "";

  data.forEach((card,cardIndex)=>{

    const div = document.createElement("div");

    div.className = "card";

    div.innerHTML = `

      <div class="header">

        <input
          class="block-title"
          value="${card.name}"
          oninput="changeBlockName(${cardIndex},this.value)"
        >

        <div class="badge">
          ${card.rows.length} Números
        </div>

      </div>

      <div class="table-wrap">

        <table>

          <thead>
            <tr>
              <th>NÚMERO</th>
              <th>STATUS</th>
              <th>OBS</th>
              <th>AÇÃO</th>
            </tr>
          </thead>

          <tbody>

            ${card.rows.map((row,rowIndex)=>`

              <tr>

                <td>
                  <input
                    value="${row.numero}"
                    oninput="updateField(${cardIndex},${rowIndex},'numero',this.value)"
                  >
                </td>

                <td>

                  <select
                    class="${row.status === 'ON' ? 'on' : 'off'}"
                    onchange="changeStatus(this,${cardIndex},${rowIndex})"
                  >

                    <option value="ON" ${row.status === 'ON' ? 'selected' : ''}>
                      ON
                    </option>

                    <option value="OFF" ${row.status === 'OFF' ? 'selected' : ''}>
                      OFF
                    </option>

                  </select>

                </td>

                <td>
                  <input
                    value="${row.observacao}"
                    oninput="updateField(${cardIndex},${rowIndex},'observacao',this.value)"
                  >
                </td>

                <td>
                  <button
                    class="delete-btn"
                    onclick="removeRow(${cardIndex},${rowIndex})"
                  >
                    X
                  </button>
                </td>

              </tr>

            `).join("")}

          </tbody>

        </table>

      </div>

      <div class="actions">

        <button onclick="addRow(${cardIndex})">
          + Adicionar Linha
        </button>

      </div>

    `;

    dashboard.appendChild(div);

  });

}

window.updateField = function(card,row,field,value){

  data[card].rows[row][field] = value;

  clearTimeout(window.saveTimer);

  window.saveTimer = setTimeout(()=>{
    saveData();
  },500);

}

window.changeStatus = function(el,card,row){

  const value = el.value;

  data[card].rows[row].status = value;

  if(value === "ON"){
    el.className = "on";
  }else{
    el.className = "off";
  }

  saveData();

}

window.changeBlockName = function(card,value){

  data[card].name = value;

  clearTimeout(window.blockTimer);

  window.blockTimer = setTimeout(()=>{
    saveData();
  },500);

}

window.addRow = function(card){

  data[card].rows.push({
    numero:"",
    status:"ON",
    observacao:""
  });

  render();

  saveData();

}

window.removeRow = function(card,row){

  data[card].rows.splice(row,1);

  render();

  saveData();

}

window.addBlock = function(){

  data.push({
    name:"Novo Bloco",
    rows:[
      {
        numero:"",
        status:"ON",
        observacao:""
      }
    ]
  });

  render();

  saveData();

}

onSnapshot(dashboardRef,(snapshot)=>{

  if(snapshot.exists()){

    const firebaseData = snapshot.data();

    data = firebaseData.cards || [];

    render();

  }else{

    saveData();

    render();

  }

});
